#if PIXELGALLERY_DEVICE_PROBE
using System.Collections;
using System.Reflection;
using UnityEngine;
using Unity.Profiling;

namespace PixelGallery
{
    public sealed class DeviceSmokeProbe : MonoBehaviour
    {
        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
        static void Launch() { new GameObject("DeviceSmokeProbe").AddComponent<DeviceSmokeProbe>(); }
        IEnumerator Start()
        {
            yield return null;
            var game = Object.FindFirstObjectByType<GalleryPrototype>();
            var flags = BindingFlags.Instance|BindingFlags.NonPublic;
            typeof(GalleryPrototype).GetField("session",flags).SetValue(game,new GallerySession(new SaveData(),d=>{}));
            typeof(GalleryPrototype).GetMethod("ReturnToAtelier",flags).Invoke(game,null);
            yield return Measure("studio");
            game.EnterDrawing(); yield return new WaitForSeconds(1);
            yield return Measure("drawing");
            for(int i=0;i<31;i++) game.Drawing.Paint(i,0,3);
            game.Evaluate(); game.StartExhibition();
            yield return Measure("exhibition");
            Debug.Log("PIXEL_GALLERY_PROBE complete; memory-only session");
        }
        IEnumerator Measure(string phase)
        {
            using(var draws=ProfilerRecorder.StartNew(ProfilerCategory.Render,"Draw Calls Count"))
            using(var triangles=ProfilerRecorder.StartNew(ProfilerCategory.Render,"Triangles Count"))
            {
                yield return new WaitForSeconds(1);
                float duration=0,maximum=0; long drawTotal=0,triangleTotal=0; int frames=0;
                while(duration<6)
                {
                    yield return null; float delta=Time.unscaledDeltaTime;
                    duration+=delta; maximum=Mathf.Max(maximum,delta); frames++;
                    if(draws.Valid)drawTotal+=draws.LastValue;
                    if(triangles.Valid)triangleTotal+=triangles.LastValue;
                }
                Debug.Log("PIXEL_GALLERY_PROBE "+phase+" frames="+frames+" seconds="+duration+" averageMs="+(duration*1000/frames)+" maxMs="+(maximum*1000)+" draws="+(drawTotal/frames)+" triangles="+(triangleTotal/frames));
            }
        }
    }
}
#endif
